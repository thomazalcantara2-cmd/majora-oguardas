(function () { 'use strict'; var state = { selectedId: null, token: null, dadosOriginais: null }; var MAX_ANEXO_BYTES = 10 * 1024 * 1024; var MIME_PERMITIDOS = [ 'application/pdf', 'image/jpeg', 'image/png', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document' ]; var els = {}; document.addEventListener('DOMContentLoaded', function () { cacheEls(); bindEvents(); }); function cacheEls() { els.cpfInput = document.getElementById('cpf-input'); els.btnBuscarCpf = document.getElementById('btn-buscar-cpf'); els.searchHint = document.getElementById('search-hint'); els.searchError = document.getElementById('search-error'); els.stepSearch = document.getElementById('step-search'); els.stepPassword = document.getElementById('step-password'); els.stepData = document.getElementById('step-data'); els.stepSuccess = document.getElementById('step-success'); els.passwordSelectedName = document.getElementById('password-selected-name'); els.passwordInput = document.getElementById('password-input'); els.passwordError = document.getElementById('password-error'); els.btnValidarSenha = document.getElementById('btn-validar-senha'); els.btnBackToSearch = document.getElementById('btn-back-to-search'); els.fMatricula = document.getElementById('f-matricula'); els.fNome = document.getElementById('f-nome'); els.fCpf = document.getElementById('f-cpf'); els.fOrdem = document.getElementById('f-ordem'); els.fDataRequerimento = document.getElementById('f-data-requerimento'); els.fStatus = document.getElementById('f-status'); els.fRequerimentoTexto = document.getElementById('f-requerimento-texto'); els.fAnexo = document.getElementById('f-anexo'); els.anexoError = document.getElementById('anexo-error'); els.btnConfirmar = document.getElementById('btn-confirmar'); els.submitError = document.getElementById('submit-error'); els.successMessage = document.getElementById('success-message'); els.successRequerimento = document.getElementById('success-requerimento'); els.loadingOverlay = document.getElementById('loading-overlay'); } function bindEvents() { els.cpfInput.addEventListener('input', function () { hideError(els.searchError); }); els.cpfInput.addEventListener('keydown', function (e) { if (e.key === 'Enter') buscarPorCpf(); }); els.btnBuscarCpf.addEventListener('click', buscarPorCpf); els.btnBackToSearch.addEventListener('click', function () { state.selectedId = null; els.passwordInput.value = ''; hideError(els.passwordError); showStep('search'); }); els.btnValidarSenha.addEventListener('click', validarSenha); els.passwordInput.addEventListener('keydown', function (e) { if (e.key === 'Enter') validarSenha(); }); els.fAnexo.addEventListener('change', function () { hideError(els.anexoError); var file = els.fAnexo.files[0]; if (!file) return; if (MIME_PERMITIDOS.indexOf(file.type) === -1) { showError(els.anexoError, 'Tipo de arquivo não permitido. Envie PDF, JPG, PNG ou DOCX.'); els.fAnexo.value = ''; return; } if (file.size > MAX_ANEXO_BYTES) { showError(els.anexoError, 'Arquivo excede o tamanho máximo permitido (10MB).'); els.fAnexo.value = ''; } }); els.btnConfirmar.addEventListener('click', enviar); } // ---------- etapa 1: identificação por CPF ---------- function buscarPorCpf() { var cpfDigitos = els.cpfInput.value.replace(/\D/g, ''); if (cpfDigitos.length !== 11) { showError(els.searchError, 'Digite os 11 números do CPF.'); return; } hideError(els.searchError); els.btnBuscarCpf.disabled = true; setLoading(true); google.script.run .withSuccessHandler(function (resp) { setLoading(false); els.btnBuscarCpf.disabled = false; if (!resp.ok) { showError(els.searchError, resp.message); return; } selecionarServidor(resp.id, resp.nome); }) .withFailureHandler(function (err) { setLoading(false); els.btnBuscarCpf.disabled = false; showError(els.searchError, 'Erro ao consultar. Tente novamente.'); }) .buscarPorCpf(cpfDigitos); } function selecionarServidor(id, nome) { state.selectedId = id; els.passwordSelectedName.textContent = nome; els.passwordInput.value = ''; hideError(els.passwordError); showStep('password'); setTimeout(function () { els.passwordInput.focus(); }, 50); } // ---------- etapa 2: senha ---------- function validarSenha() { var senha = els.passwordInput.value.trim(); if (senha.length !== 4) { showError(els.passwordError, 'Digite os 4 dígitos da senha.'); return; } hideError(els.passwordError); els.btnValidarSenha.disabled = true; setLoading(true); google.script.run .withSuccessHandler(function (resp) { setLoading(false); els.btnValidarSenha.disabled = false; if (!resp.ok) { showError(els.passwordError, resp.message); els.passwordInput.value = ''; return; } state.token = resp.token; state.dadosOriginais = resp.dados; preencherFormulario(resp.dados); showStep('data'); }) .withFailureHandler(function (err) { setLoading(false); els.btnValidarSenha.disabled = false; showError(els.passwordError, 'Erro ao validar. Tente novamente.'); }) .validarSenha(state.selectedId, senha); } function preencherFormulario(dados) { els.fMatricula.value = dados.matricula || ''; els.fNome.value = dados.nome || ''; els.fCpf.value = dados.cpfMascarado || ''; els.fOrdem.value = dados.ordem || ''; els.fDataRequerimento.value = dados.dataRequerimento || ''; els.fStatus.value = dados.status || ''; els.fRequerimentoTexto.value = ''; els.fAnexo.value = ''; hideError(els.submitError); hideError(els.anexoError); } function lerArquivoComoBase64(file) { return new Promise(function (resolve, reject) { var reader = new FileReader(); reader.onload = function () { var base64 = reader.result.split(',')[1]; resolve({ base64: base64, mimeType: file.type, filename: file.name }); }; reader.onerror = reject; reader.readAsDataURL(file); }); } function enviar() { hideError(els.submitError); if (!state.token) { showError(els.submitError, 'Sessão expirada. Refaça a identificação por CPF e a validação de senha.'); return; } var file = els.fAnexo.files[0]; if (file && (MIME_PERMITIDOS.indexOf(file.type) === -1 || file.size > MAX_ANEXO_BYTES)) { showError(els.submitError, 'Verifique o arquivo anexado antes de continuar.'); return; } setLoading(true); els.btnConfirmar.disabled = true; var montarPayloadEEnviar = function (anexoPayload) { var payload = { requerimentoTexto: els.fRequerimentoTexto.value, anexo: anexoPayload || null }; google.script.run .withSuccessHandler(function (resp) { setLoading(false); els.btnConfirmar.disabled = false; if (!resp.ok) { showError(els.submitError, resp.message || 'Não foi possível registrar. Tente novamente.'); return; } mostrarSucesso(resp); }) .withFailureHandler(function (err) { setLoading(false); els.btnConfirmar.disabled = false; showError(els.submitError, 'Erro ao registrar: ' + (err && err.message ? err.message : 'tente novamente.')); }) .registrarConfirmacao(state.token, payload); }; if (file) { lerArquivoComoBase64(file) .then(montarPayloadEEnviar) .catch(function () { setLoading(false); els.btnConfirmar.disabled = false; showError(els.submitError, 'Erro ao ler o arquivo anexado.'); }); } else { montarPayloadEEnviar(null); } } function mostrarSucesso(resp) { els.successMessage.textContent = 'Seus dados foram confirmados. Registrado em ' + resp.timestamp + '.'; if (resp.requerimentoRecebido) { els.successRequerimento.textContent = 'Seu requerimento (texto e/ou anexo) também foi recebido e está pendente de análise.'; els.successRequerimento.classList.remove('hidden'); } else { els.successRequerimento.classList.add('hidden'); } showStep('success'); } function showStep(step) { els.stepSearch.classList.toggle('hidden', step !== 'search'); els.stepPassword.classList.toggle('hidden', step !== 'password'); els.stepData.classList.toggle('hidden', step !== 'data'); els.stepSuccess.classList.toggle('hidden', step !== 'success'); } function setLoading(isLoading) { els.loadingOverlay.classList.toggle('hidden', !isLoading); } function showError(el, message) { el.textContent = message; el.classList.remove('hidden'); } function hideError(el) { el.textContent = ''; el.classList.add('hidden'); } })();
